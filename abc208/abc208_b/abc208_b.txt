#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int cluc(int n) {
    int num = 1;
    rep (i,n) num *= (i + 1);
    return num;
} 

int main() {
    int p;
    cin >> p;

    int ans = 0;
    for (int i = 10; i > 0; i--) {
        if (p < cluc(i)) {
            continue;
        }
        ans += p / cluc(i);
        p %= cluc(i);
    }
    cout << ans << endl;
}
