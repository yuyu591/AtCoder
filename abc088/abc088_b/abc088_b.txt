#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n;
    cin >> n;

    priority_queue<int> pq;
    rep(i,n) {
        int a;
        cin >> a;
        pq.push(a);
    }

    int alice = 0;
    int bob = 0;
    rep(i,n) {
        if (i % 2 == 0) alice += pq.top();
        else bob += pq.top();
        pq.pop();
    }

    cout << alice - bob << endl;
    return 0;
}
