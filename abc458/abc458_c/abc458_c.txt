#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    string s;
    cin >> s;

    int len = s.size();
    ll ans = 0;
    rep(i, s.size()) {
        if (s.at(i) == 'C') {
            int len_right = (len - 1) - i;
            int len_left = i;
            ans += min(len_right, len_left) + 1;
        }
    }
    cout << ans << endl;
    return 0;
}
