#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n,k;
    cin >> n >> k;

    ll ans = n;
    rep(i,k) {
        if (ans % 200 == 0) {
            ans /= 200;
        } else {
            ans = ans * 1000 + 200;
        }
    }

    cout << ans << endl;
}
