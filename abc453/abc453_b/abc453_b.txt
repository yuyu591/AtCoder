#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int t,x;
    cin >> t >> x;

    int save;
    cin >> save;
    cout << 0 << " " << save << endl;

    rep(i, t) {
        int a; cin >> a;
        if (abs(save - a) >= x) {
            cout << i + 1 << " " << a << endl;
            save = a;
        }
    }
    return 0;
}
