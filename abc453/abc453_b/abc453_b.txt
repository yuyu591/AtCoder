#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int t,x;
    cin >> t >> x;

    vector<int> time;
    vector<int> ans;

    int a; cin >> a;
    time.push_back(0);
    ans.push_back(a);
    int pre = a;

    rep(i, t) {
        cin >> a;
        if (abs(a - pre) >= x) {
            time.push_back(i + 1);
            ans.push_back(a);
            pre = a;
        }
    }
    
    rep(i, time.size()) {
        cout << time.at(i) << " " << ans.at(i) << endl;
    }
    return 0;
}
