#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n;
    string s;
    cin >> n >> s;

    bool flag = true;
    rep(i,n) {
        if (s.at(i) == 'o' && flag) continue;
        else {
            flag = false;
            cout << s.at(i);
        }
    }
    cout << endl;
}
