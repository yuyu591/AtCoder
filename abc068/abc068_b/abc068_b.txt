#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n;
    cin >> n;

    int max = 0;
    int ans = 1;
    rep(i, n) {
        int num = i + 1;
        int cnt = 0;
        while(num % 2 == 0) {
            num /= 2;
            cnt++;
        }
        if (cnt > max) {
            max = cnt;
            ans = i + 1;
        }
    }
    cout << ans << endl;
    return 0;
}
