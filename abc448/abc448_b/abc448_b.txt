#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n,m; cin >> n >> m;
    vector<int> c(m);
    rep(i,m) cin >> c.at(i);

    int ans = 0;
    rep(i,n) {
        int a,b; cin >> a >> b;
        if (c.at(a-1) - b >= 0) {
            ans += b;
            c.at(a-1) -= b;
        } else {
            ans += c.at(a-1);
            c.at(a-1) = 0;
        }
    }
    cout << ans << endl;
    return 0;
}
