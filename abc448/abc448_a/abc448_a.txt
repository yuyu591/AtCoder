#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n,x;
    cin >> n >> x;

    rep(i,n) {
        int a;
        cin >> a;
        if (a < x) {
            x = a;
            cout << 1 << endl;
        } else {
            cout << 0 << endl;
        }
    }
    return 0;
}
