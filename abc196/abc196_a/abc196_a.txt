#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int a,b,c,d;
    cin >> a >> b >> c >> d;

    int max = -1000;
    for(int i = a; i <= b; i++) {
        for (int k = c; k <= d; k++) {
            if ((i - k) > max) max = i - k;
        }
    }
    cout << max << endl;
    return 0;
}
