#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    int n;
    cin >> n;

    vector<vector<int>> a(n, vector<int>(n));
    int r = 0;
    int c = (n - 1) / 2;
    int k = 1;
    a.at(r).at(c) = k;
    for (int i = 0; i < n * n - 1; i++) {
        if (a.at((r - 1 + n) % n).at((c + 1) % n) == 0) {
            a.at((r - 1 + n) % n).at((c + 1) % n) = k + 1;
            r = (r - 1 + n) % n;
            c = (c + 1) % n;
            k = k + 1;
        } else {
            a.at((r + 1) % n).at(c) = k + 1;
            r = (r + 1) % n;
            k = k + 1;
        }
    }

    for (int i = 0; i < n; i++) {
        for (int k = 0; k < n; k++) {
            cout << a.at(i).at(k) << " ";
        }
        cout << endl;
    }
    return 0;
}
