#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
    ll n, m;
    cin >> n >> m;

    set<pair<int, int>> a;
    int cnt = 0;
    for (int i = 0; i < m; i++) {
        int r, c;
        cin >> r >> c;
        r--; c--;

        if (a.find({r,c}) == a.end() && a.find({r,c+1}) == a.end()
            && a.find({r+1,c}) == a.end() && a.find({r+1,c+1}) == a.end()) {
                cnt++;
                a.insert({r, c});
                a.insert({r, c+1});
                a.insert({r+1, c});
                a.insert({r+1, c+1});
        }
    }
    cout << cnt << endl;
    return 0;
}
