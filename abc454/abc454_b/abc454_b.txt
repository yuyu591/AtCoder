#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n,m;
    cin >> n >> m;

    set<int> s;
    rep(i,n) {
        int f;
        cin >> f;
        s.insert(f);
    }

    if (s.size() == n) cout << "Yes" << endl; else cout << "No" << endl;
    if (s.size() == m) cout << "Yes" << endl; else cout << "No" << endl;

    return 0;
}
