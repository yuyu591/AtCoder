#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n,m;
    cin >> n >> m;

    vector<int> v(n);
    rep(i,m) {
        int a,b;
        cin >> a >> b;

        v.at(a-1)++;
        v.at(b-1)++;
    }

    rep(i,n) cout << v.at(i) << endl;
    return 0;
}
