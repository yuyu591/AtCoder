using ll = long long;

const int INF = 1e9;

int main() {
    int n,k; cin >> n >> k;
    vector<int> h(n);
    rep(i,n) cin >> h.at(i);

    sort(all(h));

    int mini = INF;
    rep(i, h.size() - k + 1) {
        int score = h.at(i + k - 1) - h.at(i);
        mini = min(mini, score);
    }
    cout << mini << endl;
    return 0;
}
