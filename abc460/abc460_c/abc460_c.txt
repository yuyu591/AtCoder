#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n,m; cin >> n >> m;

    priority_queue<int> a,b;
    rep(i,n) {
        int A; cin >> A;
        a.push(A);
    }
    rep(i,m) {
        int B; cin >> B;
        b.push(B);
    }

    int cnt = 0;
    rep(i, m) {
        if (a.empty()) {
            break;
        }
        if (b.top() / 2.0 <= a.top()) {
            a.pop();
            cnt++;
        }
        b.pop();
    }
    cout << cnt << endl;
    return 0;
}
