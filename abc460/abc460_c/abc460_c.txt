#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n,m; cin >> n >> m;

    vector<int> a(n),b(m);
    rep(i,n) cin >> a.at(i);
    rep(i,m) cin >> b.at(i);

    sort(a.rbegin(), a.rend());
    sort(b.rbegin(), b.rend());

    int index = 0;
    int cnt = 0;
    rep(i,m) {
        if ((index < n) && (b.at(i) / 2.0 <= a.at(index))) {
            cnt++;
            index++;
        }
    }
    cout << cnt << endl;
}
