#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int h,w;
    cin >> h >> w;

    int ans = 0;
    if (h == 1 || w == 1) {
        ans = h * w;
    } else if (h % 2 == 0) {
        if (w % 2 == 0) {
            ans = (h / 2) * (w / 2);
        } else {
            ans = (h / 2) * ((w / 2) + 1);
        }
    } else {
        if (w % 2 == 0) {
            ans = ((h / 2) + 1) * (w / 2);
        } else {
            ans = ((h / 2) + 1) * ((w / 2) + 1);
        }
    }
    cout << ans << endl;
}
