#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int x;
    cin >> x;

    vector<int> v;
    rep(i,6) rep(k,6) rep(j,6) {
        v.push_back(i+1 + k+1 + j+1);
    }
    rep(i,216) {
        if(v.at(i) == x) {
            cout << "Yes" << endl;
            return 0;
        }
    }
    cout << "No" << endl;
    return 0;
}
