#include <bits/stdc++.h>
using namespace std;

int main() {
    int x,n;
    cin >> x >> n;

    vector<int> w(n), exsists(n, 0);
    for (int i = 0; i < n; i++) {
        cin >> w.at(i);
    }

    int q; cin >> q;
    int total = x;
    for (int i = 0; i < q; i++) {
        int p;
        cin >> p;

        if (exsists.at(p - 1) == 0) { //i番目の部品がついていないとき
            total += w.at(p - 1);
            cout << total << endl;
            exsists.at(p - 1) = 1;
        } else { //ついているとき
            total -= w.at(p - 1);
            cout << total << endl;
            exsists.at(p - 1) = 0;
        }
    }
    return 0;
}
