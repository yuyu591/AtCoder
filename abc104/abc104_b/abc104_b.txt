#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
#define rep1(i,n) for (int i = 1; i <= (n); i++)
using namespace std;
using ll = long long;

int main() {
    string s; cin >> s;

    rep(i, s.size()) {
        if (s.at(0) != 'A') {
            cout << "WA" << endl;
            return 0;
        }
    }

    int cnt = 0;
    for (int i = 2; i < s.size() - 1; i++) {
        if (s.at(i) == 'C') cnt++;
    }
    if (cnt != 1) {
        cout << "WA" << endl;
        return 0;
    }

    for (auto c : s) {
        if (c != 'A' && c != 'C' && isupper(c)) {
            cout << "WA" << endl;
            return 0;
        }
    }

    cout << "AC" << endl;
    return 0;
}
