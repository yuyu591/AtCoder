#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
#define rep1(i,n) for (int i = 1; i <= (n); i++)
using namespace std;
using ll = long long;

int main() {
    string s; cin >> s;

    int big = 0, small = 0;
    for (auto c : s) {
        if (isupper(c)) big++;
        else small++;
    }

    if (big > small) {
        rep(i, s.size()) {
            s.at(i) = (char)toupper(s.at(i));
        }
    } else {
        rep(i, s.size()) {
            s.at(i) = (char)tolower(s.at(i));
        }
    }
    cout << s << endl;
    return 0;
}
