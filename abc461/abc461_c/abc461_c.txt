using ll = long long;

const int INF = 1e9;

int main() {
    ll n,k,m; cin >> n >> k >> m;
    vector<pair<ll, ll>> v(n);
    rep(i,n) {
        int a,b; cin >> a >> b;
        v.at(i).first = b;
        v.at(i).second = a;
    }

    sort(v.rbegin(), v.rend());

    vector<ll> vali(n);
    ll total = 0;
    ll cnt = 0;
    rep(i,n) {
        if (cnt == k) break;
        int index = v.at(i).second - 1;
        if ((k - cnt == m) && (vali.at(index) == 1)) {
            continue;
        }

        total += v.at(i).first;
        cnt++; //選んだ宝石の数
        if (vali.at(index) == 0) {
            m--; //選ばないといけない種類
            vali.at(index) = 1;
        }
    }

    cout << total << endl;
    return 0;
}
