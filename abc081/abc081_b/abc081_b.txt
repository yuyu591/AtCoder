#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n;
    cin >> n;

    vector<int> vec(n);
    rep(i,n) cin >> vec.at(i);

    int ans = 0;
    bool ng = false;
    while(!ng) {
        rep(i,n) {
        	ng = false;
            if(vec.at(i) % 2 == 0) {
                vec.at(i) /= 2;
            } else {
                ng = true;
                break;
            }
        }
        if(!ng) ans++;
    }

    cout << ans << endl;
    return 0;
}
