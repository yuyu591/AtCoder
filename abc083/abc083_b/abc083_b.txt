#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n,a,b;
    cin >> n >> a >> b;

    int ans = 0;
    for (int num = 1; num <= n; num++) {
        int res = num;
        int sum_digit = 0;

        while(res > 0) {
            sum_digit += res % 10;
            res /= 10;
        }
        if (sum_digit >= a && sum_digit <= b) ans += num; 
    }

    cout << ans << endl;
}
