#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int h,w,Q; cin >> h >> w >> Q;
    
    rep(i,Q) {
        int q,x; cin >> q >> x;

        if (q == 1) {
            cout << x * w << endl;
            h -= x;
        } else {
            cout << x * h << endl;
            w -= x;
        }
    }
    return 0;
}
