#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n;
    cin >> n;

    vector<int> d(n);
    rep(i,n) cin >> d.at(i);
    sort(d.rbegin(), d.rend());

    int cnt = 0;
    int pre = 1000;
    rep(i,n) {
        if (d.at(i) < pre) {
            cnt++;
            pre = d.at(i);
        }
    }
    cout << cnt << endl;
    return 0;
}
