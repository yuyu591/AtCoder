#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;

int main() {
  int N, M;
  cin >> N >> M;

  vector<vector<string>> vec(N, vector<string>(N, "-"));
  rep(i, M) {
    int a,b;
    cin >> a >> b;

    vec.at(a-1).at(b-1) = "o";
    vec.at(b-1).at(a-1) = "x";
  }
  // ここにプログラムを追記
  // (ここで"試合結果の表"の2次元配列を宣言)
  rep(i, N) {
    rep(k, N) {
        cout << vec.at(i).at(k);
        if (k == N - 1) {
            cout << endl;
        } else {
            cout << " ";
        }
    }
  }
}
