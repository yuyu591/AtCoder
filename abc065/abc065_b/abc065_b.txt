#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
using ll = long long;

int main() {
    int n;
    cin >> n;

    vector<int> v(n,0);
    rep(i,n) cin >> v.at(i);

    int index = 0;
    int ans = 0;

    rep(i,n) {
        if (v.at(index) == -1) {
            ans = -1;
            break;
        }

        int next_index = v.at(index) - 1;
        if (next_index == 1) {
            ans++;
            break;
        }
        v.at(index) = -1;
        index = next_index;
        ans++;
    }
    cout << ans << endl;
    
    return 0;
}
