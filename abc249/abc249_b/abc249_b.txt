    string alpha = "abcdefghijklmnopqrstuvwxyz";
    bool flag1 = false;
    bool flag2 = false;
    bool flag3 = true;

    for (auto c : s) {
        if (isupper(c)) flag1 = true;
        else if (islower(c)) flag2 = true;
    }

    rep(i, 26) {
    	int cnt = 0;
        rep(k, s.size()) {
            if (alpha.at(i) == s.at(k)) cnt++;
        }

        }
    }
